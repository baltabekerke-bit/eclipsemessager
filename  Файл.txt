    <!-- Firebase SDK Setup & App Logic -->
    <script type="module">
        import { initializeApp } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-app.js";
        import { getAuth, signInAnonymously, signInWithCustomToken, createUserWithEmailAndPassword, signInWithEmailAndPassword, signOut, onAuthStateChanged } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-auth.js";
        import { getFirestore, doc, getDoc, setDoc, updateDoc, onSnapshot, collection, query, getDocs, addDoc } from "https://www.gstatic.com/firebasejs/11.6.1/firebase-firestore.js";

        const appId = typeof __app_id !== 'undefined' ? __app_id : 'eclipse-message-blackhole';
        const firebaseConfig = JSON.parse(__firebase_config);

        const app = initializeApp(firebaseConfig);
        const auth = getAuth(app);
        const db = getFirestore(app);

        let currentUser = null;
        let activeChatId = null;
        let unsubsMessages = null;
        let chatsCache = [];

        window.switchAuthTab = (tab) => {
            const isLogin = tab === 'login';
            document.getElementById('tab-login').className = isLogin ? 'flex-1 py-2 text-xs font-semibold rounded-lg bg-cyan-600 text-white transition' : 'flex-1 py-2 text-xs font-semibold rounded-lg text-slate-400 hover:text-white transition';
            document.getElementById('tab-register').className = !isLogin ? 'flex-1 py-2 text-xs font-semibold rounded-lg bg-cyan-600 text-white transition' : 'flex-1 py-2 text-xs font-semibold rounded-lg text-slate-400 hover:text-white transition';
            document.getElementById('name-field-container').classList.toggle('hidden', isLogin);
            document.getElementById('auth-submit-btn').textContent = isLogin ? 'Войти в систему' : 'Зарегистрировать пилота';
        };

        window.handleAuthSubmit = async (e) => {
            e.preventDefault();
            const email = document.getElementById('auth-email').value;
            const pass = document.getElementById('auth-password').value;
            const name = document.getElementById('auth-name').value;
            const isRegister = !document.getElementById('name-field-container').classList.contains('hidden');

            try {
                if (isRegister) {
                    const cred = await createUserWithEmailAndPassword(auth, email, pass);
                    await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'users', cred.user.uid), {
                        uid: cred.user.uid,
                        email: email,
                        name: name || email.split('@')[0]
                    });
                    showToast("Добро пожаловать в систему Черной Дыры!");
                } else {
                    await signInWithEmailAndPassword(auth, email, pass);
                    showToast("Успешное подключение к сингулярности!");
                }
            } catch (err) {
                showToast("Ошибка аутентификации: " + err.message);
            }
        };

        window.loginAnonymously = async () => {
            try {
                if (typeof __initial_auth_token !== 'undefined' && __initial_auth_token) {
                    await signInWithCustomToken(auth, __initial_auth_token);
                } else {
                    await signInAnonymously(auth);
                }
                const uid = auth.currentUser.uid;
                await setDoc(doc(db, 'artifacts', appId, 'public', 'data', 'users', uid), {
                    uid: uid,
                    email: 'guest@eclipse.space',
                    name: 'КосмоПилот_' + uid.substring(0, 4)
                }, { merge: true });
                showToast("Гостевой сеанс активирован!");
            } catch (err) {
                showToast("Ошибка гостевого входа: " + err.message);
            }
        };

        window.logoutUser = () => {
            signOut(auth);
        };

        onAuthStateChanged(auth, async (user) => {
            if (user) {
                currentUser = user;
                document.getElementById('auth-screen').classList.add('hidden');
                document.getElementById('main-app').classList.remove('hidden');

                const userDoc = await getDoc(doc(db, 'artifacts', appId, 'public', 'data', 'users', user.uid));
                let userName = user.email || 'Пилот';
                if (userDoc.exists()) {
                    userName = userDoc.data().name;
                }
                document.getElementById('my-display-name').textContent = userName;

                loadChats();
            } else {
                currentUser = null;
                document.getElementById('auth-screen').classList.remove('hidden');
                document.getElementById('main-app').classList.add('hidden');
            }
        });

        function loadChats() {
            const chatsRef = collection(db, 'artifacts', appId, 'public', 'data', 'chats');
            onSnapshot(chatsRef, (snapshot) => {
                chatsCache = [];
                snapshot.forEach(docSnap => {
                    const data = docSnap.data();
                    if (data.members && data.members.includes(currentUser.uid)) {
                        chatsCache.push({ id: docSnap.id, ...data });
                    }
                });
                renderChatsList();
            }, (err) => console.error(err));
        }

        window.renderChatsList = () => {
            const listContainer = document.getElementById('chats-list');
            const queryText = (document.getElementById('chat-search').value || '').toLowerCase();
            const filtered = chatsCache.filter(c => (c.title || '').toLowerCase().includes(queryText));

            if (filtered.length === 0) {
                listContainer.innerHTML = `<div class="p-6 text-center text-slate-500 text-xs">Нет активных каналов связи.</div>`;
                return;
            }

            listContainer.innerHTML = filtered.map(chat => {
                const isActive = activeChatId === chat.id;
                const title = chat.title || 'Квантовый Чат';
                return `
                    <div onclick="selectChat('${chat.id}', '${title}')" class="p-3.5 flex items-center gap-3 cursor-pointer transition hover:bg-white/5 ${isActive ? 'bg-white/10 border-l-4 border-cyan-400' : ''}">
                        <div class="w-11 h-11 rounded-full p-0.5 accretion-disk flex items-center justify-center shrink-0 shadow-md">
                            <div class="w-full h-full rounded-full bg-[#020204] flex items-center justify-center">
                                <div class="w-3.5 h-3.5 rounded-full bg-cyan-400 shadow-[0_0_8px_#38bdf8]"></div>
                            </div>
                        </div>
                        <div class="flex-1 min-w-0">
                            <h4 class="font-semibold text-sm truncate text-slate-200">${title}</h4>
                            <p class="text-xs text-slate-400 truncate">Зашифровано черной дырой</p>
                        </div>
                    </div>
                `;
            }).join('');
        };

        window.selectChat = (chatId, chatName) => {
            activeChatId = chatId;
            document.getElementById('empty-state').classList.add('hidden');
            document.getElementById('active-chat-container').classList.remove('hidden');
            document.getElementById('active-name').textContent = chatName;

            renderChatsList();
            subscribeToMessages(chatId);

            if (window.innerWidth < 768) {
                document.getElementById('sidebar').classList.add('-translate-x-full');
            }
        };

        window.closeActiveChatMobile = () => {
            document.getElementById('sidebar').classList.remove('-translate-x-full');
            document.getElementById('active-chat-container').classList.add('hidden');
            document.getElementById('empty-state').classList.remove('hidden');
            activeChatId = null;
            if (unsubsMessages) unsubsMessages();
        };

        function subscribeToMessages(chatId) {
            if (unsubsMessages) unsubsMessages();
            const msgRef = collection(db, 'artifacts', appId, 'public', 'data', 'chats', chatId, 'messages');
            
            unsubsMessages = onSnapshot(msgRef, (snapshot) => {
                const messages = [];
                snapshot.forEach(d => messages.push({ id: d.id, ...d.data() }));
                messages.sort((a, b) => (a.timestamp || 0) - (b.timestamp || 0));

                const scrollArea = document.getElementById('messages-scroll');
                scrollArea.innerHTML = messages.map(msg => {
                    const isMe = msg.senderId === currentUser.uid;
                    return `
                        <div class="flex flex-col ${isMe ? 'items-end' : 'items-start'} mb-1">
                            <div class="max-w-[75%] rounded-2xl px-4 py-2.5 text-sm shadow-md ${isMe ? 'bg-gradient-to-r from-cyan-600 to-purple-600 text-white rounded-br-none message-glow' : 'bg-[#08080e] border border-cyan-500/20 text-slate-200 rounded-bl-none'}">
                                <p class="leading-relaxed break-words">${msg.text}</p>
                                <span class="text-[10px] opacity-70 block text-right mt-1">${new Date(msg.timestamp || Date.now()).toLocaleTimeString([], {hour: '2-digit', minute:'2-digit'})}</span>
                            </div>
                        </div>
                    `;
                }).join('');
                scrollArea.scrollTop = scrollArea.scrollHeight;
            }, (err) => console.error(err));
        }

        window.handleSendMessage = async (e) => {
            e.preventDefault();
            const input = document.getElementById('message-input');
            const text = input.value.trim();
            if (!text || !activeChatId) return;

            const msgRef = collection(db, 'artifacts', appId, 'public', 'data', 'chats', activeChatId, 'messages');
            await addDoc(msgRef, {
                senderId: currentUser.uid,
                text: text,
                timestamp: Date.now()
            });

            const currentChat = chatsCache.find(c => c.id === activeChatId);
            if (currentChat && currentChat.isAi) {
                callEclipseAI(text, activeChatId);
            }

            input.value = '';
        };

        async function callEclipseAI(prompt, chatId) {
            const apiKey = "";
            const apiUrl = `https://generativelanguage.googleapis.com/v1beta/models/gemini-3-flash-preview:generateContent?key=${apiKey}`;
            try {
                const response = await fetch(apiUrl, {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({
                        contents: [{ parts: [{ text: prompt }] }],
                        systemInstruction: { parts: [{ text: "You are Eclipse AI, an intelligent entity born inside a supermassive black hole. Provide wise, mystical, yet scientifically grounded cosmic answers." }] }
                    })
                });
                const result = await response.json();
                const aiText = result.candidates?.[0]?.content?.parts?.[0]?.text || "Гравитационные помехи...";

                const msgRef = collection(db, 'artifacts', appId, 'public', 'data', 'chats', chatId, 'messages');
                await addDoc(msgRef, {
                    senderId: 'eclipse-ai-bot',
                    text: aiText,
                    timestamp: Date.now()
                });
            } catch (err) {
                console.error(err);
            }
        }

            const email = document.getElementById('new-chat-email').value.trim();
            if (!email) return;

            const usersRef = collection(db, 'artifacts', appId, 'public', 'data', 'users');
            const snapshot = await getDocs(usersRef);
            let targetUser = null;
            snapshot.forEach(d => {

            });

            if (!targetUser) {

                return;

                members: [currentUser.uid, targetUser.uid],

            });

            showToast("Канал успешно проложен!");
